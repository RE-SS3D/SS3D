# Non-admin graphical client joining an externally driven Host (e.g. the Unity editor), checking
# the UI Toolkit lobby from the remote side. Run windowed, not -batchmode (the lobby UI does not
# bootstrap headless). The Host must:
#   1. be in the lobby (Stopped, not ready) before this client connects;
#   2. start a round after the client shows Ready and has sent its chat line (client spawns on
#      Ongoing), then stop it;
#   3. start a second round without this client readying (client late-joins), then stop it.
# The client ckey must not hold Administrator permission. The Host must also perform os_* requests
# (OsClick/OsType/OsKey test signals) as real input on this window (the pilot uses Testing/ui/uinput_input.py).
wait_timeout 180
wait_connected
ui_wait #lobby-shell-surface visible
ui_wait #character-creator-surface hidden
ui_wait #lobby-ready enabled
ui_wait #lobby-admin hidden
ui_text #lobby-header-players 2/
ui_text #lobby-header-map No map
# Let the scene fade-in finish before capturing.
wait_seconds 2
screenshot remote-01-lobby

# Server-side permission check: a non-admin start request must be ignored.
probe_server_settings_permissions
start_round
wait_seconds 5
ui_text #lobby-round-status Pre-round

# Real OS input from here to the chat line: the host performs os_* requests via uinput, so hit
# testing (nothing covering the button) and keyboard focus are exercised, not just callbacks.
os_click #lobby-ready
ui_text #lobby-ready-title READY
ui_text #lobby-ready-subtitle Click to cancel
screenshot remote-02-ready

# File a report through the real dialog and broadcast transport. The host verifies the JSONL
# metadata and reduced JPEG. Reopening during cooldown keeps the new draft and disables Send.
os_click #lobby-report-bug
ui_wait #window visible
ui_text #submit SEND REPORT
os_click #summary
os_type remote pilot bug report
os_click #details
os_type reproduced from the remote client with real keyboard input.
# OS requests are serviced asynchronously; allow both strings to finish before the screenshot.
wait_seconds 6
screenshot remote-bug-report-dialog
os_click #submit
ui_wait #window hidden
ui_text #toast-text Bug report saved (round 0). Thanks!
screenshot remote-bug-report-sent
os_click #lobby-report-bug
ui_wait #window visible
os_click #summary
os_type keep this draft during cooldown
ui_wait #submit disabled
ui_text #rate-limit You can send another report in
wait_seconds 3
screenshot remote-bug-report-cooldown
os_click #cancel
ui_wait #window hidden

# Keyboard focus: click into the chat composer, type, send with Return (the host checks client.log).
os_click #tab-chat
os_click #chat-input
os_type remoteprobe real keyboard
wait_seconds 2
screenshot remote-02-chat-typed
os_key Return

# Host starts round 1: ready client spawns and the lobby hides.
wait_round Ongoing
ui_text #lobby-round-status In progress
assert_embarked
ui_wait #lobby-shell-surface hidden
screenshot remote-03-spawned

# Host stops round 1: client returns to the shell.
wait_round Stopped
ui_wait #lobby-shell-surface visible
ui_wait #character-creator-surface hidden
screenshot remote-04-returned

# Host starts round 2 without this client ready: Join opens the picker, which late-spawns.
wait_round Ongoing
ui_text #lobby-ready-title JOIN
ui_wait #lobby-ready enabled
screenshot remote-05-join-round
ui_submit #lobby-ready
# Join opens the picker (lobby effort Phase 6); its Join button spawns.
ui_wait #lobby-join-confirm enabled
screenshot remote-05b-join-picker
ui_submit #lobby-join-confirm
assert_embarked
ui_wait #lobby-shell-surface hidden

wait_round Stopped
ui_wait #lobby-shell-surface visible
screenshot remote-06-final
disconnect
